%------------------------------------------------- 4. VALUE PROPOSITIONS
\gbox{113.04}{-13.6}{51.72}{126}{tint}{\bb
\boxhead{4. Value Propositions}
\sublab{For gift buyers (paying side)}
\begin{bl}
\item Recommendations matched to the \textit{recipient}, not only to the occasion
\item Matching on interests, relationship, occasion, budget and preferences
\item Less gift-selection stress
\item Saves time: a decision in minutes rather than an evening of browsing
\item A short, focused set of options instead of endless choice
\item A clear explanation of why each gift fits that person
\item Jeddah availability and SAR pricing throughout
\item Privacy-first personalization: the 10-question quiz plus an optional opt-in recipient profile. No scraping
\item Optional reusable recipient profiles for future occasions
\item End-to-end gifting in one flow: Recommend $\rightarrow$ Buy $\rightarrow$ Wrap $\rightarrow$ Add card $\rightarrow$ Deliver \textit{(Phase 1B)}
\item Built for Saudi occasions: Eid, graduation, weddings, newborns, birthdays
\end{bl}
\vspace{0.8mm}
\sublab{For gift recipients (non-paying side)}
\begin{bl}
\item Optional opt-in Gift Profile; they review every inferred interest and can remove any of them before anything is stored
\item Nothing is collected without their own action
\end{bl}
\vspace{0.8mm}
\sublab{For retail partners (partner side)}
\begin{bl}
\item Access to buyers already showing purchase intent
\item Intent-matched referrals rather than broad advertising
\end{bl}}

%------------------------------------------------- 7. CUSTOMER SEGMENTS
\gbox{226.08}{-13.6}{51.72}{126}{tint}{\bb
\boxhead{7. Customer Segments}
\flag{GiftCompass is a multi-sided market.} Three distinct sides, only one of which pays at launch.\\[0.8mm]
\sublab{A. Gift buyers --- primary paying customers}
\begin{bl}
\item University students
\item Young professionals
\item Busy professionals with little time to shop
\item People buying for Saudi occasions: Eid, graduations, birthdays, weddings, newborns
\item Corporate gifting and HR teams --- \textit{possible future B2B segment}
\end{bl}
\vspace{0.8mm}
\sublab{B. Gift recipients --- do not pay}
\begin{bl}
\item May optionally complete their own opt-in Gift Profile to improve the match
\item Benefit without ever paying or registering
\end{bl}
\vspace{0.8mm}
\sublab{C. Retail and delivery partners --- may pay}
\begin{bl}
\item Retailers supply products and fulfil orders
\item Delivery partners carry out delivery
\item Retail partners may pay GiftCompass through commission or a commercial partnership
\end{bl}
\vspace{0.8mm}
\flag{To be validated.} These segments are assumptions. Customer interviews are
intended to test them; none has been conducted to date.}

%------------------------------------------------- 9. REVENUE STREAMS
\gbox{128.8}{-141.2}{149}{42}{bx}{\bb
\boxhead{9. Revenue Streams}
\begin{minipage}[t]{48mm}\bb
\sublab{Phase 1A --- validate demand}
\begin{bl}
\item Affiliate and referral commissions from partner retailers on referred purchases
\item Fast to test, but little control over checkout, wrapping and delivery
\end{bl}
\end{minipage}\hfill
\begin{minipage}[t]{48mm}\bb
\sublab{Phase 1B --- GiftCompass checkout}
\begin{bl}
\item Commission on partner sales under a GiftCompass-operated checkout
\item A service fee charged per order
\item A margin on gift-wrapping services
\item Greeting-card and gift-card upsells
\item Express-delivery and convenience upsells
\end{bl}
\end{minipage}\hfill
\begin{minipage}[t]{48mm}\bb
\sublab{Phase 2 --- after demand is proven}
\begin{bl}
\item Premium subscriptions, once demand is validated
\item GiftCompass-owned products in proven high-demand categories
\item Corporate gifting
\item Clearly labelled sponsored placements
\end{bl}
\end{minipage}

\vspace{0.6mm}
\flag{Trust rule.} Sponsored placements must be clearly labelled and must never silently
determine which gifts are recommended. Ranking stays driven by the recipient profile.}
